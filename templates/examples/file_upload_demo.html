使用方法 

{% verbatim %}{% include 'components/file_uploader.html' with {% endverbatim %}
    name="file"           {# 必填：字段名 #}
    label="上传文件"       {# 可选：标签 #}
    accept=".pdf,.docx"   {# 可选：允许的格式 #}
    max_size_mb=100       {# 可选：最大大小 #}
    multiple=True         {# 可选：多文件 #}
    required=True         {# 可选：是否必填 #}
    existing_file=obj.file {# 可选：已有文件 #}
    help_text="帮助文本"   {# 可选：说明文字 #}
